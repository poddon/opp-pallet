  function shuffle(a) {
    const x = a.slice();
    for (let i = x.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [x[i], x[j]] = [x[j], x[i]];
    }
    return x;
  }

    let h = 2166136261;
    const s = normG(group);
    for (let i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 16777619); }
    return "g" + (h >>> 0).toString(16) + "m" + id;
  }

      headers: { Authorization: "Bearer " + token },

  }

    const key = gkey(g, id);

    if (!keys[key]) {
      try {
        const cr = await fetch(AB + "/create/" + ABNS + "/" + key + "?initializer=" + (open ? 1 : 0));

    remoteGroup[id] = !!open;

  function canOpen(groupName, id) {
    // Общий тест (модуль 1) всегда открыт
    if (String(id) === "1") return true;
    const g = normG(groupName);
    if (loadAcc()[id]) return true;

    const asg = loadAsg().filter((x) => normG(x.group) === g && String(x.id) === String(id));
    if (asg.length) return !!asg[asg.length - 1].open;
    return false;
  }

  function prepare(id) {
    const src = (window.MODULES && MODULES[id] && MODULES[id].questions) || [];
    return shuffle(src).map((q) => {
      const order = shuffle(q.a.map((_, i) => i));
      return { q: q.q, a: order.map((i) => q.a[i]), correct: order.indexOf(q.i) };
    });
  }

  function audio() {
    if (!ac) ac = new (window.AudioContext || window.webkitAudioContext)();
    if (ac.state === "suspended") ac.resume();
    return ac;
  }
  function tone(f, d, type, g, at) {
    const c = audio();
    const o = c.createOscillator();
    const gain = c.createGain();
    o.type = type || "sine";
    o.frequency.value = f;
    gain.gain.value = g || 0.08;
    o.connect(gain); gain.connect(c.destination);
    const t0 = c.currentTime + (at || 0);
    gain.gain.setValueAtTime(g || 0.08, t0);
    gain.gain.exponentialRampToValueAtTime(0.001, t0 + d);
    o.start(t0); o.stop(t0 + d + 0.02);
  }
  function sfxOk() { tone(660, 0.08, "triangle", 0.07); tone(880, 0.12, "sine", 0.05, 0.06); }
  function sfxBad() { tone(180, 0.18, "sawtooth", 0.06); }
  function sfxSiren() {
    const c = audio();
    const o = c.createOscillator();
    const g = c.createGain();
    o.type = "sawtooth"; o.frequency.setValueAtTime(440, c.currentTime);
    o.frequency.linearRampToValueAtTime(880, c.currentTime + 0.4);
    o.frequency.linearRampToValueAtTime(440, c.currentTime + 0.8);
    g.gain.value = 0.04; o.connect(g); g.connect(c.destination);
    o.start(); o.stop(c.currentTime + 1.2);
  }

    const ok = choice === q.correct;
    [...$("opts").children].forEach((b, i) => {
      if (i === q.correct) b.classList.add("ok");
      else if (i === choice) b.classList.add("bad");
      else b.classList.add("dim");
      b.disabled = true;
    });
    $("verdict").textContent = ok ? "Верно" : (choice < 0 ? "Время" : "Неверно");
    $("verdict").className = "verdict " + (ok ? "ok" : "bad");

      const bonus = Math.max(8, Math.round(left / TMAX * 20));
      streak += 1; correctN += 1; xp += 12 + bonus + (streak >= 3 ? streak * 2 : 0);
    } else {
      sfxBad(); if (window.OPP_FACTORY) { OPP_FACTORY.cue = "drop"; OPP_FACTORY.cueUntil = performance.now() + 900; }
      streak = 0;
    }
    setTimeout(() => {
      if (idx + 1 >= qs.length) finish("Пройден");
      else { idx += 1; locked = false; left = TMAX; renderQ(); timer = setInterval(tick, 1000); }
    }, 1150);
  }
